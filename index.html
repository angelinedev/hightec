💡 6. Palindrome Checker 


function isPalindrome(str) {
  // Normalize string: lowercase and remove non-alphanumeric characters
  str = str.toLowerCase().replace(/[^a-z0-9]/g, ''); 
  return str === str.split('').reverse().join('');
}
console.log(isPalindrome("madam")); // true
console.log(isPalindrome("A man, a plan, a canal: Panama")); // true

🔁 8. FizzBuzz (Classic) 


// Prints "Fizz" for multiples of 3, "Buzz" for multiples of 5, 
// "FizzBuzz" for multiples of both, and the number otherwise.
for (let i = 1; i <= 15; i++) {
  let out = "";
  if (i % 3 === 0) out += "Fizz";
  if (i % 5 === 0) out += "Buzz";
  console.log(out || i); // If 'out' is empty, print 'i'
}
